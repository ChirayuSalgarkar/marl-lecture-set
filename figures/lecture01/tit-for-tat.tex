\begin{tikzpicture}[font=\small]
  % state machine
  \begin{scope}[shorten >=1pt,node distance=2.8cm,on grid,auto,every state/.style={minimum size=1.2cm,font=\small}]
    \node[state,initial,draw=cgreen,fill=cgreen!15,initial text={start}] (C) at (0,0) {play C};
    \node[state,draw=cred,fill=cred!15] (D) [right=of C] {play D};
    \path[->,thick]
      (C) edge [loop above,cgreen] node {\scriptsize other: C} ()
          edge [bend left,cred] node {\scriptsize other: D} (D)
      (D) edge [loop above,cred] node {\scriptsize other: D} ()
          edge [bend left,cgreen] node {\scriptsize other: C} (C);
  \end{scope}
  % example play
  \begin{scope}[shift={(5.6,-0.2)}]
    \node[anchor=east,font=\scriptsize] at (-0.1,0.8) {other};
    \node[anchor=east,font=\scriptsize] at (-0.1,-0.2) {Tit-for-Tat};
    \foreach \o/\m [count=\k from 0] in {C/C,C/C,D/C,C/D,D/C,D/D,C/D}{
      \if\o C\def\co{cgreen}\else\def\co{cred}\fi
      \if\m C\def\cm{cgreen}\else\def\cm{cred}\fi
      \node[circle,draw=\co,fill=\co!20,minimum size=0.55cm,inner sep=0,font=\scriptsize] (o\k) at (\k*0.85+0.3,0.8) {\o};
      \node[circle,draw=\cm,fill=\cm!20,minimum size=0.55cm,inner sep=0,font=\scriptsize] (m\k) at (\k*0.85+0.3,-0.2) {\m};
      \node[font=\tiny,text=cgray] at (\k*0.85+0.3,-0.75) {\k};}
    \foreach \k [evaluate=\k as \n using int(\k+1)] in {0,...,5}
      \draw[->,cgray] (o\k) -- (m\n);
    \node[note] at (3.0,-1.2) {time step $t$};
  \end{scope}
\end{tikzpicture}%
